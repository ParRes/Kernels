#
# CMake port of MPIRMA/*/Makefile. Same structure as SERIAL/CMakeLists.txt
# (see that file for the general pattern this mirrors): each kernel lives
# in its own subdirectory with its own Makefile including
# ../../common/MPI.defs -- CCOMPILER=$(MPICC), COMOBJS=MPI_bail_out.o
# wtime.o, PROG_ENV=-DMPI. These kernels use MPI one-sided (RMA) calls
# (MPI_Win/MPI_Get/MPI_Put etc.) via ordinary MPI API calls in the .c
# sources -- no separate CMake dependency beyond find_package(MPI).
#
# Standalone project: `cd MPIRMA && cmake -S . -B build` works on its own
# (in-source `cmake .` too). Also add_subdirectory()'d from the repo root.
#

cmake_minimum_required(VERSION 3.18)
project(PRK_MPIRMA LANGUAGES C)
enable_testing()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)
find_package(MPI REQUIRED COMPONENTS C)

set(PRK_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../include")
set(PRK_COMMON_DIR "${CMAKE_CURRENT_LIST_DIR}/../common")

add_library(prk_mpirma_wtime OBJECT "${PRK_COMMON_DIR}/wtime.c")
target_include_directories(prk_mpirma_wtime PRIVATE ${PRK_INCLUDE_DIR})
target_compile_definitions(prk_mpirma_wtime PRIVATE MPI)
target_link_libraries(prk_mpirma_wtime PRIVATE MPI::MPI_C)

add_library(prk_mpirma_bail_out OBJECT "${PRK_COMMON_DIR}/MPI_bail_out.c")
target_include_directories(prk_mpirma_bail_out PRIVATE ${PRK_INCLUDE_DIR})
target_compile_definitions(prk_mpirma_bail_out PRIVATE MPI)
target_link_libraries(prk_mpirma_bail_out PRIVATE MPI::MPI_C)

# prk_mpirma_executable(<name> DIR <subdir> SOURCES <files...> [DEFS ...] [LIBS ...])
# Mirrors the shared bits of MPIRMA/*/Makefile: PROG_ENV=-DMPI, links
# COMOBJS (MPI_bail_out.o wtime.o), include path covers both this repo's
# include/ and the kernel's own subdirectory (for generated .incl files).
function(prk_mpirma_executable name)
  set(oneValueArgs DIR)
  set(multiValueArgs SOURCES DEFS LIBS)
  cmake_parse_arguments(PME "" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})
  set(_srcs "")
  foreach(_src IN LISTS PME_SOURCES)
    list(APPEND _srcs "${PME_DIR}/${_src}")
  endforeach()
  set(_target mpirma_${name})
  add_executable(${_target} ${_srcs} $<TARGET_OBJECTS:prk_mpirma_wtime> $<TARGET_OBJECTS:prk_mpirma_bail_out>)
  set_target_properties(${_target} PROPERTIES OUTPUT_NAME ${name})
  target_include_directories(${_target} PRIVATE ${PRK_INCLUDE_DIR} "${CMAKE_CURRENT_LIST_DIR}/${PME_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/${PME_DIR}")
  target_compile_definitions(${_target} PRIVATE MPI ${PME_DEFS})
  target_link_libraries(${_target} PRIVATE Threads::Threads MPI::MPI_C ${PME_LIBS})
endfunction()

function(prk_mpirma_test test_name)
  set(multiValueArgs COMMAND)
  cmake_parse_arguments(PMT "" "" "${multiValueArgs}" ${ARGN})
  add_test(NAME ${test_name} COMMAND ${PMT_COMMAND})
  set_tests_properties(${test_name} PROPERTIES TIMEOUT 30)
endfunction()

# --- Synch_p2p --------------------------------------------------------
# VERBOSE=0, RESTRICT_KEYWORD=0
prk_mpirma_executable(p2p DIR Synch_p2p SOURCES p2p.c
  DEFS VERBOSE=0 RESTRICT_KEYWORD=0)

# --- Stencil ------------------------------------------------------------
# DOUBLE=1, STAR=1, RESTRICT_KEYWORD=0, RADIUS=2, LOOPGEN=0, VERBOSE=0.
# stencil.c #includes "loop_body_star.incl"/"loop_body_compact.incl"
# unconditionally (both code paths compiled in, selected at runtime) --
# these aren't checked in, generated by common/Stencil/loop_gen at build
# time, same as the Makefile's own pattern rule (and SERIAL's own Stencil
# target).
set(_stencil_gen_dir "${CMAKE_CURRENT_BINARY_DIR}/Stencil")
file(MAKE_DIRECTORY ${_stencil_gen_dir})
add_custom_command(
  OUTPUT "${_stencil_gen_dir}/loop_body_star.incl"
  COMMAND "${PRK_COMMON_DIR}/Stencil/loop_gen" 2 1
  WORKING_DIRECTORY ${_stencil_gen_dir}
  COMMENT "Generating Stencil/loop_body_star.incl (RADIUS=2)")
add_custom_command(
  OUTPUT "${_stencil_gen_dir}/loop_body_compact.incl"
  COMMAND "${PRK_COMMON_DIR}/Stencil/loop_gen" 2 0
  WORKING_DIRECTORY ${_stencil_gen_dir}
  COMMENT "Generating Stencil/loop_body_compact.incl (RADIUS=2)")
prk_mpirma_executable(stencil DIR Stencil SOURCES stencil.c
  DEFS DOUBLE=1 STAR=1 RESTRICT_KEYWORD=0 RADIUS=2 LOOPGEN=0 VERBOSE=0
  LIBS m)
target_sources(mpirma_stencil PRIVATE "${_stencil_gen_dir}/loop_body_star.incl" "${_stencil_gen_dir}/loop_body_compact.incl")
set_source_files_properties("${_stencil_gen_dir}/loop_body_star.incl" "${_stencil_gen_dir}/loop_body_compact.incl" PROPERTIES HEADER_FILE_ONLY TRUE)

# --- Transpose ------------------------------------------------------------
prk_mpirma_executable(transpose DIR Transpose SOURCES transpose.c
  DEFS VERBOSE=0 RESTRICT_KEYWORD=0)
