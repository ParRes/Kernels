#
# CMake port of SERIAL/*/Makefile. Unlike FORTRAN/C1z/Cxx11 (flat source
# directories), each SERIAL kernel lives in its own subdirectory with its
# own Makefile including ../../common/SERIAL.defs -- CCOMPILER=$(CC),
# COMOBJS=wtime.o, PROG_ENV=-DSERIAL. This port mirrors that: one
# executable per kernel subdirectory, common/wtime.c (+ random_draw.c for
# PIC) compiled once and linked into all of them, and each kernel's own
# -D tuning flags matching its Makefile's own `ifndef ... default` values
# (there are no #ifndef fallbacks in the .c sources themselves -- they
# rely on always receiving these defines from the command line).
#
# Standalone project: `cd SERIAL && cmake -S . -B build` works on its own
# (in-source `cmake .` too). Also add_subdirectory()'d from the repo root.
#

cmake_minimum_required(VERSION 3.18)
project(PRK_SERIAL LANGUAGES C)
enable_testing()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)

set(PRK_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../include")
set(PRK_COMMON_DIR "${CMAKE_CURRENT_LIST_DIR}/../common")

add_library(prk_serial_wtime OBJECT "${PRK_COMMON_DIR}/wtime.c")
target_include_directories(prk_serial_wtime PRIVATE ${PRK_INCLUDE_DIR})
target_compile_definitions(prk_serial_wtime PRIVATE PRK_SERIAL)

add_library(prk_serial_random_draw OBJECT "${PRK_COMMON_DIR}/random_draw.c")
target_include_directories(prk_serial_random_draw PRIVATE ${PRK_INCLUDE_DIR})
target_compile_definitions(prk_serial_random_draw PRIVATE PRK_SERIAL)

# prk_serial_executable(<name> DIR <subdir> SOURCES <files...> [DEFS ...] [LIBS ...])
# Mirrors the shared bits of SERIAL/*/Makefile: PROG_ENV=-DSERIAL, links
# COMOBJS (wtime.o), include path covers both this repo's include/ and the
# kernel's own subdirectory (for its own headers and any generated .incl
# files placed in the matching build subdirectory).
#
# The CMake *target* is named serial_<name> (e.g. serial_p2p), not <name>:
# every one of PRK's language directories has its own p2p/stencil/
# transpose/etc., and when this file is add_subdirectory()'d from the root
# alongside them, CMake target names must be unique across the *entire*
# combined project, not just within this file (add_executable(p2p ...) in
# two different subdirectories is a hard configure error: "another target
# with the same name already exists"). OUTPUT_NAME is set back to the bare
# <name> so the actual produced binary and every prk_serial_test() COMMAND
# below (which refer to the binary by its on-disk name, not the CMake
# target name) are unaffected.
function(prk_serial_executable name)
  set(oneValueArgs DIR)
  set(multiValueArgs SOURCES DEFS LIBS)
  cmake_parse_arguments(PSE "" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})
  set(_srcs "")
  foreach(_src IN LISTS PSE_SOURCES)
    list(APPEND _srcs "${PSE_DIR}/${_src}")
  endforeach()
  set(_target serial_${name})
  add_executable(${_target} ${_srcs} $<TARGET_OBJECTS:prk_serial_wtime>)
  set_target_properties(${_target} PROPERTIES OUTPUT_NAME ${name})
  target_include_directories(${_target} PRIVATE ${PRK_INCLUDE_DIR} "${CMAKE_CURRENT_LIST_DIR}/${PSE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/${PSE_DIR}")
  target_compile_definitions(${_target} PRIVATE PRK_SERIAL ${PSE_DEFS})
  target_link_libraries(${_target} PRIVATE Threads::Threads ${PSE_LIBS})
endfunction()

function(prk_serial_test test_name)
  set(multiValueArgs COMMAND)
  cmake_parse_arguments(PST "" "" "${multiValueArgs}" ${ARGN})
  add_test(NAME ${test_name} COMMAND ${PST_COMMAND})
  set_tests_properties(${test_name} PROPERTIES TIMEOUT 30)
endfunction()

# --- Synch_p2p --------------------------------------------------------
# VERBOSE=0, RESTRICT_KEYWORD=0
prk_serial_executable(p2p DIR Synch_p2p SOURCES p2p.c
  DEFS VERBOSE=0 RESTRICT_KEYWORD=0)

# --- Stencil ------------------------------------------------------------
# DOUBLE=1, STAR=1, RESTRICT_KEYWORD=0, RADIUS=2, LOOPGEN=0, VERBOSE=0.
# stencil.c #includes "loop_body_star.incl"/"loop_body_compact.incl"
# unconditionally (both code paths compiled in, selected at runtime) --
# these aren't checked in, generated by common/Stencil/loop_gen at build
# time, same as the Makefile's own pattern rule.
set(_stencil_gen_dir "${CMAKE_CURRENT_BINARY_DIR}/Stencil")
file(MAKE_DIRECTORY ${_stencil_gen_dir})
add_custom_command(
  OUTPUT "${_stencil_gen_dir}/loop_body_star.incl"
  COMMAND "${PRK_COMMON_DIR}/Stencil/loop_gen" 2 1
  WORKING_DIRECTORY ${_stencil_gen_dir}
  COMMENT "Generating Stencil/loop_body_star.incl (RADIUS=2)")
add_custom_command(
  OUTPUT "${_stencil_gen_dir}/loop_body_compact.incl"
  COMMAND "${PRK_COMMON_DIR}/Stencil/loop_gen" 2 0
  WORKING_DIRECTORY ${_stencil_gen_dir}
  COMMENT "Generating Stencil/loop_body_compact.incl (RADIUS=2)")
prk_serial_executable(stencil DIR Stencil SOURCES stencil.c
  DEFS DOUBLE=1 STAR=1 RESTRICT_KEYWORD=0 RADIUS=2 LOOPGEN=0 VERBOSE=0)
target_sources(serial_stencil PRIVATE "${_stencil_gen_dir}/loop_body_star.incl" "${_stencil_gen_dir}/loop_body_compact.incl")
set_source_files_properties("${_stencil_gen_dir}/loop_body_star.incl" "${_stencil_gen_dir}/loop_body_compact.incl" PROPERTIES HEADER_FILE_ONLY TRUE)

# --- Branch (not part of ci/build-run-prk.sh's allserial coverage, but a
# normal kernel with its own subdirectory like the rest) -------------------
# branch.c doesn't #include func.c -- it's a separate translation unit
# (matches the Makefile's OBJS = $(PROGRAM).o $(COMOBJS) func.o), generated
# by the checked-in ./func_gen script (MATRIX_RANK=10, NUMBER_OF_FUNCTIONS=40
# defaults).
set(_branch_gen_dir "${CMAKE_CURRENT_BINARY_DIR}/Branch")
file(MAKE_DIRECTORY ${_branch_gen_dir})
add_custom_command(
  OUTPUT "${_branch_gen_dir}/func.c"
  COMMAND "${CMAKE_CURRENT_LIST_DIR}/Branch/func_gen" 10 40
  WORKING_DIRECTORY ${_branch_gen_dir}
  COMMENT "Generating Branch/func.c (MATRIX_RANK=10, NUMBER_OF_FUNCTIONS=40)")
add_executable(serial_branch Branch/branch.c "${_branch_gen_dir}/func.c" $<TARGET_OBJECTS:prk_serial_wtime>)
set_target_properties(serial_branch PROPERTIES OUTPUT_NAME branch)
target_include_directories(serial_branch PRIVATE ${PRK_INCLUDE_DIR} "${CMAKE_CURRENT_LIST_DIR}/Branch")
target_compile_definitions(serial_branch PRIVATE PRK_SERIAL VERBOSE=0 RESTRICT_KEYWORD=0)
target_link_libraries(serial_branch PRIVATE Threads::Threads)
